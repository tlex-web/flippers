**Analog:** case (c) in the same file, lines 362-388. Use the same per-target loop, the `makeTableWorld()` (:50-56) helper, and the raise/return stepping:
```ts
for (const target of [0, 1]) {
  const world = makeTableWorld();
  const sample = (): void => {
    const f = world.getFlippers().find((x) => x.id === target);
    if (!f) throw new Error(`no flipper ${target}`);
    ...
  };
  sample();
  world.setFlipperMotor(target, 1);
  for (let i = 0; i < 30; i++) { world.step(FIXED_DT); sample(); }

  for (let i = 0; i < 40; i++) { world.step(FIXED_DT); sample(); }
```
